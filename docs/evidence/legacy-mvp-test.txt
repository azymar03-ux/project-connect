import {test,expect} from '@playwright/test';
test('discover, demo login, save, message, create and apply',async({page})=>{
 await page.goto('http://localhost:3000');await expect(page.locator('.project-card')).toHaveCount(6);
 await page.locator('.sign-in').click();await page.getByRole('button',{name:'Explore with the demo account'}).click();await expect(page.locator('.sidebar-profile')).toContainText('Nivas Kumar');
 if(!await page.locator('.save-button').first().evaluate(e=>e.classList.contains('saved')))await page.locator('.save-button').first().click();await page.locator('.sidebar').getByRole('button',{name:'Saved projects'}).click();await expect(page.locator('.project-card').first()).toBeVisible();
 await page.locator('.sidebar').getByRole('button',{name:'Messages',exact:true}).click();await page.locator('.chat-person').first().click();await page.getByRole('textbox',{name:'Message',exact:true}).fill('Hello! Excited to build together.');await page.getByRole('button',{name:'Send message'}).click();await expect(page.locator('.message-bubble').last()).toContainText('Excited to build');
 await page.locator('.sidebar').getByRole('button',{name:'Discover',exact:true}).click();await page.getByRole('button',{name:'Create project',exact:true}).click();await page.locator('[name=title]').fill('Test maker community project');await page.locator('textarea[name=description]').fill('A real end-to-end test project for curious builders and makers.');await page.locator('[name=skills]').fill('Python, Robotics');await page.locator('[name=positions]').fill('Python Developer');await page.getByRole('button',{name:'Publish project'}).click();await expect(page.locator('.project-card')).toHaveCount(2);
 await page.locator('.sidebar').getByRole('button',{name:'Discover',exact:true}).click();await page.getByRole('button',{name:'Smart Energy Monitor',exact:true}).click();await page.getByRole('button',{name:'I’d love to build this'}).click();await page.getByRole('textbox',{name:'Say hello to the team'}).fill('I can contribute Python and IoT experience to this project.');await page.getByRole('button',{name:'Send join request'}).click();await expect(page.locator('.member-banner')).toContainText('Application pending');
});
test('mobile discovery has no horizontal overflow',async({page})=>{await page.setViewportSize({width:390,height:844});await page.goto('http://localhost:3000');await expect(page.locator('.project-card')).toHaveCount(7);expect(await page.evaluate(()=>document.documentElement.scrollWidth<=window.innerWidth)).toBeTruthy();await page.screenshot({path:'screenshots/mobile.png',fullPage:true});});
